# Versions are pinned exactly, not floored with >=.
#
# Why: the model artifacts in inference/capstone_model_v2/ are pickles, and
# pickles are tied to the library version that wrote them. With >= ranges, a
# rebuild on a new host resolves to whatever is newest that day, so the same
# commit can produce a different image next week and the models may load
# wrong -- or silently mispredict -- with no code change to blame.
#
# These are the versions currently running and passing the test suite.
# To upgrade, bump deliberately and re-run the tests.

flask-cors==6.0.5
gunicorn==26.0.0

# Imported directly by api_server.py, so pinned as first-class deps rather
# than left to float as transitive ones. numpy especially: the 1.x -> 2.x
# break changes pickle compatibility.
numpy==2.5.1
pillow==12.3.0
pandas==3.0.3
scipy==1.18.0

# Model stack. See the note above about pickle/version coupling.
